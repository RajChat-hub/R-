cmake_minimum_required(VERSION 3.20)
project(UNFALTERING VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Only Linux epoll is implemented; kqueue and IOCP are not supported yet")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(UNFALTERING_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(UNFALTERING_FUZZ "Build libFuzzer HTTP target (Clang only)" OFF)
add_library(uf_core src/http/http.cpp src/config/config.cpp src/static/static.cpp src/routing/routing.cpp src/net/server.cpp)
target_include_directories(uf_core PUBLIC include)
target_compile_options(uf_core PUBLIC -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
if(UNFALTERING_SANITIZERS)
 target_compile_options(uf_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
 target_link_options(uf_core PUBLIC -fsanitize=address,undefined -no-pie)
endif()
add_executable(unfaltering src/cli/main.cpp)
target_link_libraries(unfaltering PRIVATE uf_core)
include(CTest)
if(BUILD_TESTING)
 add_executable(uf_unit tests/unit/tests.cpp)
 target_link_libraries(uf_unit PRIVATE uf_core)
 add_test(NAME unit COMMAND uf_unit)
 add_executable(uf_parser_stress tests/unit/parser_stress.cpp)
 target_link_libraries(uf_parser_stress PRIVATE uf_core)
 add_test(NAME parser_stress COMMAND uf_parser_stress)
 add_test(NAME cli_version COMMAND unfaltering --version)
 add_test(NAME cli_help COMMAND unfaltering --help)
 add_test(NAME example_config COMMAND unfaltering --check --config ${CMAKE_CURRENT_SOURCE_DIR}/examples/server.ufc)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_test(NAME network_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/http_test.py $<TARGET_FILE:unfaltering>)
 set_tests_properties(network_integration PROPERTIES TIMEOUT 60)
endif()
if(UNFALTERING_FUZZ)
 if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "Fuzz target requires Clang")
 endif()
 add_executable(uf_fuzz_http tests/fuzz/http_fuzz.cpp src/http/http.cpp)
 target_include_directories(uf_fuzz_http PRIVATE include)
 target_compile_options(uf_fuzz_http PRIVATE -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
 target_link_options(uf_fuzz_http PRIVATE -fsanitize=fuzzer,address,undefined)
 add_executable(uf_fuzz_config tests/fuzz/config_fuzz.cpp src/config/config.cpp src/http/http.cpp)
 target_include_directories(uf_fuzz_config PRIVATE include)
 target_compile_options(uf_fuzz_config PRIVATE -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
 target_link_options(uf_fuzz_config PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
install(TARGETS unfaltering RUNTIME DESTINATION bin)
