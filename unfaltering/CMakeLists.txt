cmake_minimum_required(VERSION 3.20)
project(UNFALTERING VERSION 0.3.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Only Linux epoll is implemented; kqueue and IOCP are not supported yet")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Only 64-bit Linux is supported by the current wire-counter/resource model")
endif()
option(UNFALTERING_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(UNFALTERING_TSAN "Enable ThreadSanitizer separately from ASan" OFF)
if(UNFALTERING_TSAN AND UNFALTERING_SANITIZERS)
 message(FATAL_ERROR "TSan and ASan require separate builds")
endif()
option(UNFALTERING_FUZZ "Build libFuzzer HTTP target (Clang only)" OFF)
add_library(uf_core src/http/http.cpp src/http/stream.cpp src/config/config.cpp src/static/static.cpp src/routing/routing.cpp src/net/server.cpp src/core/io_pool.cpp src/proxy/headers.cpp src/proxy/native.cpp src/tls/tls.cpp)
find_package(Threads REQUIRED)
find_package(OpenSSL 3.0 REQUIRED COMPONENTS SSL Crypto)
message(STATUS "TLS dependency: OpenSSL ${OPENSSL_VERSION}")
target_link_libraries(uf_core PUBLIC Threads::Threads OpenSSL::SSL OpenSSL::Crypto)
target_include_directories(uf_core PUBLIC include)
target_compile_options(uf_core PUBLIC -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
if(UNFALTERING_SANITIZERS)
 target_compile_options(uf_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
 target_link_options(uf_core PUBLIC -fsanitize=address,undefined -no-pie)
endif()
if(UNFALTERING_TSAN)
 target_compile_options(uf_core PUBLIC -fsanitize=thread -fno-omit-frame-pointer -fno-pie)
 target_link_options(uf_core PUBLIC -fsanitize=thread -no-pie)
endif()
add_executable(unfaltering src/cli/main.cpp)
target_link_libraries(unfaltering PRIVATE uf_core)
include(CTest)
if(BUILD_TESTING)
 add_executable(uf_concurrency tests/unit/concurrency_tests.cpp)
 target_link_libraries(uf_concurrency PRIVATE uf_core)
 add_test(NAME concurrency_unit COMMAND uf_concurrency)
 add_executable(uf_unit tests/unit/tests.cpp)
 target_link_libraries(uf_unit PRIVATE uf_core)
 add_test(NAME unit COMMAND uf_unit)
 add_executable(uf_stream_unit tests/unit/stream_tests.cpp)
 target_link_libraries(uf_stream_unit PRIVATE uf_core)
 add_test(NAME streaming_unit COMMAND uf_stream_unit)
 add_executable(uf_proxy_unit tests/unit/proxy_parser_tests.cpp)
 target_link_libraries(uf_proxy_unit PRIVATE uf_core)
 add_test(NAME proxy_parser_unit COMMAND uf_proxy_unit)
 add_executable(uf_parser_stress tests/unit/parser_stress.cpp)
 target_link_libraries(uf_parser_stress PRIVATE uf_core)
 add_test(NAME parser_stress COMMAND uf_parser_stress)
 add_test(NAME cli_version COMMAND unfaltering --version)
 add_test(NAME cli_help COMMAND unfaltering --help)
 add_test(NAME example_config COMMAND unfaltering --check --config ${CMAKE_CURRENT_SOURCE_DIR}/examples/server.ufc)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_test(NAME network_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/http_test.py $<TARGET_FILE:unfaltering>)
 add_test(NAME hardening_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/hardening_test.py $<TARGET_FILE:unfaltering>)
 add_test(NAME proxy_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/proxy_test.py $<TARGET_FILE:unfaltering>)
 set_tests_properties(network_integration hardening_integration PROPERTIES TIMEOUT 60)
 add_test(NAME proxy_demo COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/demo_test.py $<TARGET_FILE:unfaltering> ${CMAKE_CURRENT_SOURCE_DIR}/examples)
 set_tests_properties(proxy_demo PROPERTIES TIMEOUT 30)
 add_executable(uf_tls_io tests/unit/tls_io_tests.cpp)
 target_link_libraries(uf_tls_io PRIVATE uf_core)
 add_test(NAME tls_partial_io COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/tls_io_test.py $<TARGET_FILE:uf_tls_io>)
 set_tests_properties(tls_partial_io PROPERTIES TIMEOUT 30)
 add_test(NAME tls_demo COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/tls_demo_test.py $<TARGET_FILE:unfaltering> ${CMAKE_CURRENT_SOURCE_DIR}/examples)
 set_tests_properties(tls_demo PROPERTIES TIMEOUT 30)
 add_test(NAME tls_sustained_concurrency COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/tls_soak_test.py $<TARGET_FILE:unfaltering>)
 set_tests_properties(tls_sustained_concurrency PROPERTIES TIMEOUT 180)
 add_test(NAME tls_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/tls_test.py $<TARGET_FILE:unfaltering>)
 set_tests_properties(tls_integration PROPERTIES TIMEOUT 180)
 add_test(NAME sustained_concurrency COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/soak_test.py $<TARGET_FILE:unfaltering>)
 set_tests_properties(sustained_concurrency PROPERTIES TIMEOUT 120)
 set_tests_properties(proxy_integration PROPERTIES TIMEOUT 120)
endif()
if(UNFALTERING_FUZZ)
 if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "Fuzz target requires Clang")
 endif()
 add_executable(uf_fuzz_http tests/fuzz/http_fuzz.cpp src/http/http.cpp src/http/stream.cpp)
 target_include_directories(uf_fuzz_http PRIVATE include)
 target_compile_options(uf_fuzz_http PRIVATE -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
 target_link_options(uf_fuzz_http PRIVATE -fsanitize=fuzzer,address,undefined)
 add_executable(uf_fuzz_response tests/fuzz/response_fuzz.cpp src/http/http.cpp src/http/stream.cpp)
 target_include_directories(uf_fuzz_response PRIVATE include)
 target_compile_options(uf_fuzz_response PRIVATE -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
 target_link_options(uf_fuzz_response PRIVATE -fsanitize=fuzzer,address,undefined)
 add_executable(uf_fuzz_config tests/fuzz/config_fuzz.cpp src/config/config.cpp src/http/http.cpp src/http/stream.cpp)
 target_include_directories(uf_fuzz_config PRIVATE include)
 target_compile_options(uf_fuzz_config PRIVATE -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
 target_link_options(uf_fuzz_config PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
install(TARGETS unfaltering RUNTIME DESTINATION bin)
